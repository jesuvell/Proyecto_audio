bash# 1. Inicializa el repositorio local
git init

# 2. Añade todos los archivos (el .gitignore excluirá lo innecesario)
git add .

# 3. Guarda los cambios localmente
git commit -m "Primer commit: Estructura del proyecto y requirements"

# 4. Cambia el nombre de la rama principal a 'main'
git branch -M main

# 5. Vincula tu repositorio local con el de GitHub
# (Reemplaza la URL con la de tu repositorio vacío en GitHub)
git remote add origin https://github.com

# 6. Sube los archivos
git push -u origin main
